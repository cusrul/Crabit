cmake_minimum_required(VERSION 3.20)
project(nanoui LANGUAGES CXX)
if(CMAKE_PROJECT_NAME STREQUAL PROJECT_NAME)
    include(CTest)
endif()
option(NANOUI_BUILD_TESTS "Build original NanoUI contract tests" ${BUILD_TESTING})
option(NANOUI_DYNAMIC_STORAGE "Use standard dynamic storage rather than injected pool limits" ON)
option(NANOUI_BUILD_EXPERIMENTAL_TESTS "Build preserved, non-product experimental framework tests" OFF)
set(NANOUI_BUILD_EXAMPLES OFF)

add_library(nanoui_product STATIC src/NanoUI/product/runtime.cpp)
add_library(NanoUI::core ALIAS nanoui_product)
target_include_directories(nanoui_product PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
set_target_properties(nanoui_product PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(NANOUI_DYNAMIC_STORAGE)
    target_compile_definitions(nanoui_product PUBLIC NANOUI_DYNAMIC_STORAGE=1)
else()
    target_compile_definitions(nanoui_product PUBLIC NANOUI_DYNAMIC_STORAGE=0)
endif()

if(NANOUI_BUILD_TESTS)
    add_executable(nanoui_utf8 tests/utf8_tests.cpp)
    target_link_libraries(nanoui_utf8 PRIVATE NanoUI::core)
    set_target_properties(nanoui_utf8 PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
    add_test(NAME nanoui_utf8 COMMAND nanoui_utf8)
    foreach(profile IN ITEMS pool dynamic)
        if(profile STREQUAL "pool")
            set(storage 0)
        else()
            set(storage 1)
        endif()
        add_library(nanoui_test_${profile}_runtime STATIC src/NanoUI/product/runtime.cpp)
        target_include_directories(nanoui_test_${profile}_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
        target_compile_definitions(nanoui_test_${profile}_runtime PUBLIC NANOUI_DYNAMIC_STORAGE=${storage})
        set_target_properties(nanoui_test_${profile}_runtime PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_executable(nanoui_${profile}_navigation tests/unified_navigation_tests.cpp tests/font_stub.cpp)
        target_link_libraries(nanoui_${profile}_navigation PRIVATE nanoui_test_${profile}_runtime)
        set_target_properties(nanoui_${profile}_navigation PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_test(NAME nanoui_${profile}_navigation COMMAND nanoui_${profile}_navigation)
        add_executable(nanoui_${profile}_shutdown tests/shutdown_tests.cpp tests/font_stub.cpp)
        target_link_libraries(nanoui_${profile}_shutdown PRIVATE nanoui_test_${profile}_runtime)
        set_target_properties(nanoui_${profile}_shutdown PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_test(NAME nanoui_${profile}_shutdown COMMAND nanoui_${profile}_shutdown)
        add_executable(nanoui_${profile}_popup tests/popup_tests.cpp tests/font_stub.cpp)
        target_link_libraries(nanoui_${profile}_popup PRIVATE nanoui_test_${profile}_runtime)
        set_target_properties(nanoui_${profile}_popup PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_test(NAME nanoui_${profile}_popup COMMAND nanoui_${profile}_popup)
        add_executable(nanoui_${profile}_widgets tests/original_widgets_tests.cpp tests/font_stub.cpp)
        target_include_directories(nanoui_${profile}_widgets PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/NanoUI/product")
        target_link_libraries(nanoui_${profile}_widgets PRIVATE nanoui_test_${profile}_runtime)
        set_target_properties(nanoui_${profile}_widgets PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_test(NAME nanoui_${profile}_widgets COMMAND nanoui_${profile}_widgets)
        add_executable(nanoui_${profile}_callback_components tests/callback_components_tests.cpp tests/font_stub.cpp)
        target_link_libraries(nanoui_${profile}_callback_components PRIVATE nanoui_test_${profile}_runtime)
        set_target_properties(nanoui_${profile}_callback_components PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_test(NAME nanoui_${profile}_callback_components COMMAND nanoui_${profile}_callback_components)
        add_executable(nanoui_${profile}_container_compatibility tests/container_compatibility_tests.cpp tests/font_stub.cpp)
        target_link_libraries(nanoui_${profile}_container_compatibility PRIVATE nanoui_test_${profile}_runtime)
        set_target_properties(nanoui_${profile}_container_compatibility PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_test(NAME nanoui_${profile}_container_compatibility COMMAND nanoui_${profile}_container_compatibility)
        add_executable(nanoui_${profile}_layout tests/layout_primitives_tests.cpp tests/font_stub.cpp)
        target_link_libraries(nanoui_${profile}_layout PRIVATE nanoui_test_${profile}_runtime)
        set_target_properties(nanoui_${profile}_layout PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_test(NAME nanoui_${profile}_layout COMMAND nanoui_${profile}_layout)
        add_executable(nanoui_${profile}_flex tests/flex_compatibility_tests.cpp tests/font_stub.cpp)
        target_link_libraries(nanoui_${profile}_flex PRIVATE nanoui_test_${profile}_runtime)
        set_target_properties(nanoui_${profile}_flex PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
        add_test(NAME nanoui_${profile}_flex COMMAND nanoui_${profile}_flex)
        foreach(contract IN ITEMS leaf overlay core frame macro asset)
            add_executable(nanoui_${profile}_layout_${contract}
                tests/layout_${contract}_contract_tests.cpp tests/font_stub.cpp)
            target_link_libraries(nanoui_${profile}_layout_${contract} PRIVATE nanoui_test_${profile}_runtime)
            set_target_properties(nanoui_${profile}_layout_${contract} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
            add_test(NAME nanoui_${profile}_layout_${contract} COMMAND nanoui_${profile}_layout_${contract})
        endforeach()
    endforeach()
endif()

    include(experimental/host-tests.cmake)
